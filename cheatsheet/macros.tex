\def\navigatrix{
	\coordinate (c) at (5,5);
	\def\wb{0.5}
	\def\L{4.5}
	\foreach \theta/\set in {
		0/{l/{next char}, e/{next word end}, w/{next word start},{\small)}/{sentence end},L/{end line}},
		90/{k/{up line},<C-u>/{1/2 up},K/{scroll top}, {\small \{}/{para start}, {gg}/{first line}},     
		180/{h/{prev. char}, b/{prev. word start},  {\small(}/{sentence start}, H/{first char.},$\vert$/{start line}},
		270/{j/{down line}, <C-d>/{1/2 down}, J/{screen bottom}, {\small\}}/{para end}, G/{last line}}}
	{
		\def\col{gray!40!red!50!white}
		\def\col{green!50!black}
		\fill[fill=\col,rotate=\theta] (c) + (0,\wb) rectangle ++(\L,-\wb)--++(0,-\wb)--++(\wb,2*\wb)--++(-\wb,2*\wb)--++(0,-\wb);
		\foreach[count= \c] \char/\desc in \set
		{
			\def\r{0.85}
			\node[anchor=north,color=white] at ($(c) + ({\c*\r*cos(\theta)},{\c*\r*sin(\theta) +\wb})$) {\parbox[t]{1cm}{\centering \tt \char}};
			\node[anchor=north,color=white] at ($(c) + ({\c*\r*cos(\theta)},{\c*\r*sin(\theta) +\wb-0.35})$) {\parbox[t]{0.8cm}{\centering \tiny \desc}};
		}
	}

\def\grammar
{
	\node[fill=operator!90!white,anchor = north west] at (5.6,9.7){
		\begin{minipage}{10cm}
			
        {\bf\centering \normalsize  {Operators \& Ranges}\par }
		
		\scriptsize 
		
		Operators perform an action on \textbf{range}, which is defined by an (optional) count and a motion.
		$$ \text{\texttt{cmd} = [operator] \textcolor{gray}{[count (1)]} [motion]}$$
		The operator action is performed on all characters between the current cursor location and the location defined by (count)$\times$(motion). 
		
		
			{\bf  \small {The Operators}\par}
			
			\tab[1.6cm]
			{
				\element{y (yank)}{Copy region to registry}
				\element{d (delete)}{Copy region to registry and delete the region}
				\element{c (change)}{Copy region to registry, delete the region, and enter insert mode}
				\element{</> (indent)}{Decrease/Increase the indent level of the region}
				\element{gu/gU}{Change region to lower/upper case text}
			}

		\begin{minipage}[t]{4cm}
			\phantom{This text is invisible but holds the space.}
		\end{minipage}%
		\begin{minipage}[t]{6cm} %some hacky manual spacing to go around the navigatrix
			\phantom{i} If no count is specified, the motion is performed \\ \phantom{a}~once. Counts may preceed the operator; 5yy \\ \phantom{i}~~~and y5y are equivalent. 
			
			\phantom{a}~Repeating the operator command applies it to \\ \phantom{i} the entire current line (\texttt{dd} deletes the entire line)
		\end{minipage}


		{\bf  \small {Operator Grammar }\par}
		
		Operator-motions work like navigation:: \texttt{yW} yanks everything until the end of the current WORD, and \texttt{ctj} changes all text up to (but not including) the first `j'.  There are also two special `adverbs' which can be used after operators: \texttt{a} and \texttt{i}.

		Some keys keep their meaning (\texttt{iw} is `inside the current word'), but others change drastically:

		\vspace{0.05cm}

		\begin{minipage}[t]{5cm}
			
		\tab{
				\element{a\char}{`Around': All text specified by \char including the delimiters: \texttt{daw} deletes the current word, and the surrounding whitespace.}
				\element{i\char }{`Inside': As with \texttt{a}, but only the contents; \texttt{diw} would delete only the word.}
				\element{s\char}{Activates \href{https://github.com/tpope/vim-surround}{surround} extension. }
				}
			\end{minipage}~\begin{minipage}[t]{5cm}
				\tab{
					\element{s}{Current Sentence}
					\element{p}{Current Paragraph}
					\element{(, )}{Parenthesis group}
					\element{\{, \}}{Brace group}
					\element{[,]}{Bracket group}
					\element{<,>}{Chevron group}
					\element{t}{Tag group}
					\element{",'}{Quoted string}
				}
				% \vspace{0.2cm}
			\end{minipage}

		\end{minipage}

	};
}

\def\register
{
	\node[fill = red!30!white,anchor=north] at ($(multi.south) + (0,-0.05)$) {
		\begin{minipage}{6cm}
			{\bf\centering \normalsize Registers\par}
		
			\scriptsize \texttt{d, y} and \texttt{p} transfer text into a \textbf{register}. If none is specified, \texttt{""} is used, otherwise commands can use a specific register; \texttt{"ayy} yanks to the \texttt{a} register, and \texttt{"ap} would then paste it. 

			Some registers are assigned automatically:
			\tab{
				\element{"/}{Last search}
				\element{"0}{Last yank}
				\element{"1-"9}{Delete carousel ("1 is most recent delete, "2 second most recent, etc.)}
				\element{"+}{System clipboard (mapped to "" by default)}
			}

			Macros occupy the same registry space as \texttt{dyp}. Marks occupy a different registry space, with \texttt{a-z} local to the file, \texttt{A-Z} shared globally.
			\vspace{-0.12cm}
		\end{minipage}
	};
